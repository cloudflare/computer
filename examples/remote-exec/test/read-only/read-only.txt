the workspace holds notes, written by an agent that may write
%require
===
echo "the parent's notes" | workspace put notes.txt
---

===
a read-only agent's commands run as nobody, and can read pushed files but not change them
===
workspace run-read <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  const { exitCode, stdout, stderr } = await exec(
    ["sh", "-c", "id -u; id -g; stat -c '%U %a' notes.txt; cat notes.txt; echo changed > notes.txt"],
    { push: ["notes.txt"] },
  );
  return { exitCode, stdout, stderr };
}
EOF
workspace get notes.txt
---
{
  "exitCode": 2,
  "stdout": "65534\n65534\nroot 644\nthe parent's notes\n",
  "stderr": "sh: 1: cannot create notes.txt: Permission denied\n"
}
the parent's notes

===
a read-only agent's commands keep no capabilities, and can't gain any
===
workspace run-read <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  const { stdout } = await exec(["grep", "-E", "^(CapPrm|CapEff|CapAmb|NoNewPrivs):", "/proc/self/status"]);
  return stdout.trim().split("\n");
}
EOF
---
[
  "CapPrm:\t0000000000000000",
  "CapEff:\t0000000000000000",
  "CapAmb:\t0000000000000000",
  "NoNewPrivs:\t1"
]

===
a read-only agent can't pull, or choose who its commands run as
===
workspace run-read <<'EOF'
import { exec } from "codemode:exec";
import { exec as spawn } from "codemode:remote-exec";

const refusal = (promise) => promise.then(() => "ran", (error) => error.message);

export default async function () {
  return {
    pull: await refusal(exec(["touch", "made.txt"], { pull: ["made.txt"] })),
    user: await refusal(exec(["id", "-u"], { user: "0:0" })),
    streamingPull: await refusal(spawn(["touch", "made.txt"], { pull: ["made.txt"] })),
  };
}
EOF
workspace get made.txt
---
{
  "pull": "exec: this agent can only read files, so it can't pull files back into the workspace",
  "user": "exec: this agent can only read files, so its commands run as nobody (65534:65534) and it can't choose a user",
  "streamingPull": "exec: this agent can only read files, so it can't pull files back into the workspace"
}
(missing)
