===
the workspace holds a small project
%require
===
printf 'node_modules/\n*.log\n' | workspace put .gitignore
printf 'console.log("hello from", process.argv[2] ?? "nobody");\n' | workspace put app/src/main.js
printf 'one\ntwo\nthree\n' | workspace put app/src/data.txt
echo "ignored by .gitignore" | workspace put app/debug.log
echo "ignored too" | workspace put app/node_modules/dep/index.js
echo "the command deletes me" | workspace put app/stale.txt
---

===
exec pushes the files, runs the command and pulls back what it changed
%require
===
workspace run <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  return await exec(
    [
      "sh",
      "-c",
      [
        "find app -type f | sort",
        "node app/src/main.js codemode",
        "mkdir -p app/out && wc -l < app/src/data.txt > app/out/count.txt",
        "tr a-z A-Z < app/src/data.txt > app/src/data.tmp && mv app/src/data.tmp app/src/data.txt",
        "rm app/stale.txt",
        "mkdir -p app/node_modules/new && echo x > app/node_modules/new/index.js",
        "echo to-stderr >&2",
      ].join(" && "),
    ],
    { cwd: "/workspace", push: ["app/**", ".gitignore"], pull: ["app/**"] },
  );
}
EOF

===
a second push sends only the files that changed
===
echo "changed in the workspace" | workspace put app/src/data.txt
workspace run <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  const { stdout, sync } = await exec(["cat", "app/src/data.txt"], { push: ["app/**"] });
  return { stdout, pushed: sync.pushed };
}
EOF

===
paths in push and pull are relative to cwd
===
workspace run <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  const { stdout, sync } = await exec(["sh", "-c", "ls; echo made > made.txt"], {
    cwd: "/workspace/app/src",
    push: ["*.js"],
    pull: ["*.txt"],
  });
  return { stdout, pushed: sync.pushed.files, pulled: sync.pulled.files };
}
EOF
workspace get app/src/made.txt
---
{
  "stdout": "data.txt\nmain.js\n",
  "pushed": 1,
  "pulled": 1
}
made

===
a command can run as another user, who owns the files pushed for it
===
echo "mine" | workspace put u/file.txt
workspace run <<'EOF'
import { exec } from "codemode:exec";

export default async function () {
  const { stdout } = await exec(
    ["sh", "-c", "id -u; id -g; stat -c %u:%g u u/file.txt; touch u/made.txt"],
    { user: "1000:1000", push: ["u/**"], pull: ["u/**"] },
  );
  return stdout.trim().split("\n");
}
EOF
workspace get u/made.txt
---
[
  "1000",
  "1000",
  "1000:1000",
  "1000:1000"
]
